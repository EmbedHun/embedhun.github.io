1. 개요 및 배경

임베디드 제어 시스템의 기능이 복잡해짐에 따라 단일 무한 루프(Super Loop) 방식의 펌웨어 설계는 작업 간 지연, 통신 패킷 처리 지연 등 타이밍 신뢰성 저하 문제에 맞닥뜨립니다.

이를 해결하기 위해 멀티태스킹과 우선순위 기반 선점형 스케줄링을 지원하는 실시간 운영체제(RTOS)의 적용이 적극적으로 검토됩니다.

특히 Keil MDK 환경에서 제공되는 CMSIS-RTX RTOS는 뛰어난 컴팩트함과 아주 작은 메모리 풋프린트(Footprint) 덕분에 자원이 부족한 8비트/32비트 마이크로컨트롤러에 이상적입니다.

그중에서도 누보톤 Cortex-M0 코어 기반 베스트셀러 모델인 NUC140 시리즈 학습 보드 및 개발 보드 환경에서 RTX RTOS를 구동하려는 실무 요구가 많습니다.

이 가이드북은 누보톤 표준 개발 생태계 내에서 Keil RTX RTOS 커널을 NUC140 보드에 안전하게 포팅하고 하드웨어 타이머 시스템(SysTick)을 동기화하여 멀티태스킹 태스크를 기동하는 펌웨어 개발 시퀀스를 제공합니다.

2. RTX RTOS 구성 요소 라이브러리 준비

포팅을 시작하기 전 Keil MDK 내에 포함된 CMSIS 라이브러리 파일들을 타겟 누보톤 프로젝트 경로에 포함시켜야 합니다.

필수 파일 구성 리스트

- RTX_Conf_CM.c: RTX RTOS 커널 세부 설정 파일 (태스크 개수, 스택 크기 등 정의)

- RTX_CM0.lib: Cortex-M0 코어 전용으로 사전 컴파일된 RTX 라이브러리 이진 파일

- CMSIS_RTX.h: RTX API 정의 공용 헤더 파일

3. SysTick 및 하드웨어 벡터 리디렉션 설정

RTX RTOS는 태스크 스케줄링을 제어하기 위해 Cortex-M0 내부에 탑재된 시스템 틱 타이머(SysTick Timer)의 1ms 주기 하드웨어 인터럽트를 활용합니다.

포팅 단계

1. 기존 누보톤 스타트업 파일(startup_NUC100Series.s)의 인터럽트 핸들러 매핑에서 기본 SysTick_Handler 인터럽트 함수가 RTX 커널의 스케줄러 진입점 함수로 분기되도록 리디렉션 처리를 수행해야 합니다.

2. C 소스코드 내의 기존 SysTick_Handler() 사용자 선언을 완벽히 제거합니다. (RTX 라이브러리 내부에서 이 핸들러를 물리적으로 점유하여 사용하기 때문입니다.)

[NUC140 메인 멀티태스킹 구동 C 소스코드]

#include <stdio.h>

#include "NuMicro.h"

#include "cmsis_os.h" // CMSIS-RTX API 포함

// 태스크 ID 선언

osThreadId tid_Thread1;

osThreadId tid_Thread2;

// 태스크 1 함수 선언

void Thread1(void const *argument)

{

while(1)

{

// LED 토글 제어 등

printf("Task 1 구동 중\n");

osDelay(500); // 500ms 동안 대기 상태(Blocked)로 천이하여 스케줄러로 제어권 반환

}

}

// 태스크 2 함수 선언

void Thread2(void const *argument)

{

while(1)

{

printf("Task 2 구동 중\n");

osDelay(1000); // 1000ms 동안 대기 상태로 천이

}

}

// 태스크 속성 스펙 정의

osThreadDef(Thread1, osPriorityNormal, 1, 0);

osThreadDef(Thread2, osPriorityNormal, 1, 0);

int main(void)

{

SYS_UnlockReg();

// 1. 하드웨어 클럭 소스 설정 (내장 22.1184MHz HIRC 사용 및 시스템 50MHz 동작)

CLK_EnableXtalRC(CLK_PWRCTL_HIRCEN_Msk);

CLK_WaitClockReady(CLK_STATUS_HIRCSTB_Msk);

CLK_SetHCLK(CLK_CLKSEL0_HCLKSEL_HIRC, CLK_CLKDIV0_HCLK(1));

// 2. UART 디버그 포트 초기화

CLK_EnableModuleClock(UART0_MODULE);

CLK_SetModuleClock(UART0_MODULE, CLK_CLKSEL1_UARTSEL_HIRC, CLK_CLKDIV0_UART(1));

SYS->GPB_MFPL = (SYS->GPB_MFPL & ~SYS_GPB_MFPL_PB0MFP_Msk) | SYS_GPB_MFPL_PB0MFP_UART0_RXD;

SYS->GPB_MFPL = (SYS->GPB_MFPL & ~SYS_GPB_MFPL_PB1MFP_Msk) | SYS_GPB_MFPL_PB1MFP_UART0_TXD;

UART_Open(UART0, 115200);

SYS_LockReg();

printf("RTX RTOS 초기화 기동...\n");

// 3. RTX OS 커널 초기화

osKernelInitialize();

// 4. 각각의 독립된 실행 태스크 생성

tid_Thread1 = osThreadCreate(osThread(Thread1), NULL);

tid_Thread2 = osThreadCreate(osThread(Thread2), NULL);

// 5. OS 스케줄러 기동 시작 (이 단계 이후 메인 루프는 중지되고 RTOS 태스크 제어로 넘어감)

osKernelStart();

while(1)

{

// 여기까지 실행이 넘어오지 않음

}

}

4. 임베디드 FAE 리스크 분석 및 예방 대책

태스크별 스택 오버플로우에 따른 Hard Fault 리스크

Cortex-M0 MCU인 NUC140은 가용한 내부 SRAM이 단 8KB~16KB 수준으로 극도로 부족합니다. 만약 RTX_Conf_CM.c 설정에서 기본 개별 태스크 스택 크기(Default Stack Size)를 필요 이상으로 크게(예: 1KB 이상) 지정하고 다수의 태스크를 띄우면, 즉각 가용 SRAM 용량을 초과하여 메모리 오버플로우를 일으킵니다. 반대로 너무 작게(예: 200바이트 이하) 주면, 태스크 내부에서 지역 변수 선언이나 printf 연산 수행 시 스택 범위를 초과하여 하드폴트(Hard Fault)를 유발합니다. 반드시 개 개별 태스크의 스택 크기를 256~512바이트 범위에서 미세 조정하며 시스템 안정성을 확보하십시오.

인터럽트 내 RTOS API 사용 금지 예외

NUC140의 UART나 ADC 외부 하드웨어 인터럽트 서비스 루틴(ISR) 내부에서 RTOS 일반 API(예: osDelay 등)를 호출하면 커널 대기 행(Hang) 상태에 걸려 즉각 락업이 걸립니다. 인터럽트 핸들러 내부에서는 오직 ISR 전용 세마포어나 메일박스 트리거 API만 사용하도록 펌웨어 로직을 통제해야 합니다.
