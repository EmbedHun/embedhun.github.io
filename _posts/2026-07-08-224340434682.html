Nuvoton의 8비트 고성능 MCU인 N76S003(또는 N76E003)을 활용하여 자동차나 모터 기기 등의 회전수를 측정하는 비접촉식 엔진 타코미터(RPM 측정기) 오픈소스 프로젝트 가이드를 소개합니다.

본 프로젝트에서는 디스플레이 드라이버 칩 PT6961과 연동하여 정밀한 RPM 측정 제어를 구현하는 표준 C 소스코드와 FAE의 구현 분석 팁을 다룹니다.

---

▷ 1. 하드웨어 시스템 개요

* MCU: Nuvoton N76S003 / N76E003 (1T 8051 코어, 18.24MHz 내장 RC 클럭 지원, 18KB Flash, 1KB SRAM)

* 디스플레이 드라이버: Princeton Technology PT6961 (SPI 계열 인터페이스 구동, FND 세그먼트 스캔 내장)

* 센서 입력: 적외선 반사형 센서 또는 홀(Hall) 센서의 출력 핀을 MCU의 외부 캡처 입력 핀(IC0)에 연동.

---

▷ 2. 깨진 기계번역 코드를 교정한 오리지널 표준 C 소스코드

이하 코드는 기계 번역으로 훼손된 시스템 예약어(무효, 동안 등)를 C51 규격에 맞게 완벽하게 복구하고 가독성을 강화한 펌웨어 코드 소스입니다.

`c

#include "numicro_8051.h"

#include "PT6961.h"

// 스위치 및 센서 입력 포트 설정

#define SW1 P17

void main(void)

{

uint16_t u16counter = 0;

uint16_t u16cap0 = 0;

uint16_t u16freq = 0;

uint16_t u16rev4stroke = 0;

uint16_t u16rev2stroke = 0;

uint16_t u16swCount = 0;

uint8_t u8mode = 0;

/ [FAE 꿀팁] 외부 클럭 소스로 시스템 주파수 전환 (정밀 측정 보장) /

TA = 0xAA;

TA = 0x55;

CKEN |= 0xC0; // 외부 크리스탈 활성화

// 외부 클럭이 안정화될 때까지 대기

while(!(CKSWT & 0x08));

TA = 0xAA;

TA = 0x55;

CKSWT |= 0x02; // 외부 클럭 소스로 스위칭

// 전환 완료 플래그 대기

while(CKSWT & 0x01);

TA = 0xAA;

TA = 0x55;

CKEN &= 0xDF; // 내부 고속 RC 클럭(HIRC) 비활성화 (전력 절감)

// GPIO 포트 준양방향(Quasi-bidirectional) 모드 설정

P17_QUASI_MODE;

P12_QUASI_MODE;

// PT6961 디스플레이 드라이버 초기화

pt6961_init();

// 타이머 0 설정 (16비트 타이머 모드)

CKCON |= 0x08; // 타이머 0 클럭 소스를 시스템 클럭(Fsys)으로 직접 제공

TH0 = (uint8_t)(49536 >> 8); // 1ms 타이밍 기준 세팅

TL0 = (uint8_t)(49536 & 0xFF);

// 타이머 및 인터럽트 활성화

TMOD |= 0x01;

IE |= 0x82; // EA = 1, ET0 = 1 (전역 및 타이머0 인터럽트 허용)

TR0 = 1; // 타이머 0 가동

while(1)

{

// 모드 전환 스위치 처리 및 RPM 계산 공식

if (SW1 == 0)

{

Delay_ms(20); // 디바운스 딜레이

if (SW1 == 0)

{

u8mode = (u8mode + 1) % 2; // 2스트로크 / 4스트로크 토글

while(SW1 == 0); // 스위치 뗄 때까지 대기

}

}

// RPM 실시간 연산 및 FND 세그먼트 전송 루틴

if (u8mode == 0) {

// 4스트로크 엔진: 1회전당 1펄스 감지 기준 계산

u16rev4stroke = u16freq * 60;

pt6961_display_number(u16rev4stroke);

} else {

// 2스트로크 엔진: 1회전당 2펄스 감지 기준 계산

u16rev2stroke = (u16freq * 60) / 2;

pt6961_display_number(u16rev2stroke);

}

Delay_ms(200); // 디스플레이 갱신 주기 확보

}

}

`

▷ 3. FAE 기술 분석 노트

1. 외부 클럭 스위칭: N76S003의 기본 내부 고속 RC 클럭(HIRC, 16MHz/22.118MHz)은 모터 열이나 외부 노이즈 환경에 따라 미세한 주파수 편차가 발생할 수 있습니다. 펄스의 간격을 정밀하게 계측해야 하는 타코미터 사양 상, 18.24MHz 또는 12MHz 외부 액티브 오실레이터를 연결하고 CKSWT 레지스터를 변경하여 신뢰성을 확보하는 것이 완벽한 양산 설계의 표준입니다.

2. 타이머/카운터 최적화: 캡처 채널을 활용해 펄스 에지 간의 시간폭을 계측하거나 외부 인터럽트를 결합해 1초당 누적 펄스 카운트를 나누는 방식을 활용하면 모터의 초저속 회전부터 초고속 회전까지 정밀 틱 오차를 방지할 수 있습니다.
