1. 개요 및 배경

Keil MDK 의 CMSIS-RTOS RTX 는 Cortex-M 프로세서 기반 디바이스용 범용 RTOS 인터페이스인 CMSIS-RTOS API 를 구현한 RTOS 로, 스레드 우선순위와 선택형 라운드로빈(Round-Robin) 스레드 전환을 지원합니다.

이 글은 누보톤 Cortex-M0 코어 기반 NUC140 시리즈 보드 환경을 대상으로 합니다.

이 가이드북은 Keil RTX RTOS 커널을 NUC140 보드에 포팅하고, SysTick 을 RTX 커널 타이머로 사용하여 멀티태스킹 태스크를 기동하는 펌웨어 작성 순서를 설명합니다.

2. RTX RTOS 구성 요소 라이브러리 준비

포팅에는 아래 CMSIS-RTOS RTX 파일들이 필요합니다.

필수 파일 구성 리스트

  • RTX_Conf_CM.c: RTX RTOS 커널 세부 설정 파일 (태스크 개수, 스택 크기 등 정의)
  • RTX_CM0.lib: Cortex-M0 코어 전용으로 사전 컴파일된 RTX 라이브러리 이진 파일
  • cmsis_os.h: CMSIS-RTOS API 정의 공용 헤더 파일

3. SysTick 핸들러 설정

RTX RTOS는 태스크 스케줄링을 제어하기 위해 Cortex-M0 내부에 탑재된 시스템 틱 타이머(SysTick Timer)의 1ms 주기 하드웨어 인터럽트를 활용합니다.

포팅 단계

  1. 누보톤 스타트업 파일(startup_NUC100Series.s)의 SysTick_Handler 는 WEAK 로 정의되어 있어, RTX 라이브러리가 제공하는 SysTick_Handler 가 링크 시 이를 대체합니다. 스타트업 파일에 별도의 리디렉션 코드를 넣을 필요는 없습니다.
  2. RTX 라이브러리(Cortex-M0 용)가 SVC_Handler, PendSV_Handler, SysTick_Handler 를 직접 정의하므로, C 소스코드에 사용자가 정의한 SysTick_Handler() 가 있으면 제거합니다.

[NUC140 메인 멀티태스킹 구동 C 소스코드]

#include <stdio.h>
#include "NUC100Series.h"
#include "cmsis_os.h" // CMSIS-RTX API 포함
// 태스크 ID 선언
osThreadId tid_Thread1;
osThreadId tid_Thread2;
// 태스크 1 함수 선언

void Thread1(void const *argument)
{
    while(1)
    {
        // LED 토글 제어 등
        printf("Task 1 구동 중\n");
        osDelay(500); // 500ms 대기
    }
}
// 태스크 2 함수 선언

void Thread2(void const *argument)
{
    while(1)
    {
        printf("Task 2 구동 중\n");
        osDelay(1000); // 1000ms 대기
    }
}
// 태스크 속성 스펙 정의
osThreadDef(Thread1, osPriorityNormal, 1, 0);
osThreadDef(Thread2, osPriorityNormal, 1, 0);

int main(void)
{
    SYS_UnlockReg();
    // 1. 하드웨어 클럭 소스 설정 (내장 22.1184MHz HIRC 를 HCLK 로 사용, 시스템 22.1184MHz 동작)
    CLK_EnableXtalRC(CLK_PWRCON_OSC22M_EN_Msk);
    CLK_WaitClockReady(CLK_CLKSTATUS_OSC22M_STB_Msk);
    CLK_SetHCLK(CLK_CLKSEL0_HCLK_S_HIRC, CLK_CLKDIV_HCLK(1));
    // 2. UART 디버그 포트 초기화
    CLK_EnableModuleClock(UART0_MODULE);
    CLK_SetModuleClock(UART0_MODULE, CLK_CLKSEL1_UART_S_HIRC, CLK_CLKDIV_UART(1));
    SYS->GPB_MFP &= ~(SYS_GPB_MFP_PB0_Msk | SYS_GPB_MFP_PB1_Msk);
    SYS->GPB_MFP |= (SYS_GPB_MFP_PB0_UART0_RXD | SYS_GPB_MFP_PB1_UART0_TXD);
    UART_Open(UART0, 115200);
    SYS_LockReg();
    printf("RTX RTOS 초기화 기동...\n");
    // 3. RTX OS 커널 초기화
    osKernelInitialize();
    // 4. 각각의 독립된 실행 태스크 생성
    tid_Thread1 = osThreadCreate(osThread(Thread1), NULL);
    tid_Thread2 = osThreadCreate(osThread(Thread2), NULL);
    // 5. OS 스케줄러 기동 시작 (osKernelStart 이후에도 main 함수는 메인 스레드로 계속 실행됨)
    osKernelStart();
    while(1)
    {
        // osKernelStart 이후 main 스레드가 계속 실행하는 루프
    }
}

4. 임베디드 FAE 리스크 분석 및 예방 대책

태스크별 스택 오버플로우 리스크

Cortex-M0 MCU인 NUC140은 가용한 내부 SRAM이 4KB~16KB 수준(모델에 따라 4KB/8KB/16KB)입니다. RTX_Conf_CM.c 의 기본 스레드 스택 크기(Default Thread stack size)는 64~4096바이트 범위에서 설정하며, 기본값은 200바이트입니다. 스택 오버플로우 검사(Check for stack overflow, OS_STKCHECK) 옵션을 켜면 스택 검사 코드가 포함되며, 실행 중인 스레드의 스택 오버플로우가 감지되면 os_error() 함수가 OS_ERROR_STACK_OVF 코드로 호출됩니다.

인터럽트 내 RTOS API 사용 금지 예외

NUC140의 UART나 ADC 외부 하드웨어 인터럽트 서비스 루틴(ISR) 내부에서 RTOS 일반 API(예: osDelay 등)를 호출하면 osErrorISR 오류 코드가 반환되며, 구현에 따라 HardFault 로 이어질 수 있습니다. 인터럽트 핸들러 내부에서는 ISR 에서 호출이 허용된 API(예: osSignalSet, osSemaphoreRelease, osMessagePut, osMailPut)만 사용하도록 펌웨어 로직을 통제해야 합니다.
